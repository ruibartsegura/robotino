    while (exercise_1) {
      // Señal mando parar ejecucion
      if (IrReceiver.decodedIRData.decodedRawData == 0x2DFA40) {
        exercise_1 = false;
        exercise_2 = false;
        stop();
      }

      if (millis() - ej1_start_time >= EJ1_TIEMPO) {
        exercise_1 = false;
        stop();
        digitalWrite(LED_R, LOW);
        digitalWrite(LED_B, LOW);
        break;
      }

      obj_det.distL = ultrasoundL.get_dist();
      float dist_lat = ultrasoundLat.get_dist();

      if (!follow_wall) {
        stop();
        if (obj_det.distL <= 15) {
          girar_90_dch();
          follow_wall = true;
        } else if (dist_lat <= 15) {
          follow_wall = true;
        } else {
          girar_90_dch();
          girar_90_dch();
        }
      } else {
          if (dist_lat < 5) {
            servo_right.write(0);
            servo_left.write(0);
          } else if (dist_lat > 15) {
            servo_right.write(180);
            servo_left.write(180);
          } else {
            avanzar_recto();
          }
      }
    }
